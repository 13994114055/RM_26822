# MirocFly 下位机 BetaFPV G473 (INAV 10.0.0) 对接配置 — 上位机 MSP + 物理 CRSF + MTF-02P 光流
# 用法: FC 插 USB -> picocom/Configurator 进 CLI -> 逐行粘贴 -> save (或整段 batch 粘贴)
# 前提: 拔桨! 详见 ../connection/README.md
#
# 端口号口径(io/serial.h): USART1=0 USART2=1 USART3=2 USART4=3 ; USB_VCP=20
# 函数位: MSP=1 ; RX_SERIAL=64
#
# --- 端口 ---
# 注意: INAV 限制最多 3 个 MSP 端口 (MAX_MSP_PORT_COUNT=3)。
# 默认 VCP + USART1 已占 2 个; 我们要 USART2(上位机)+USART4(光流) 也用 MSP => 必须先清空 USART1。
serial 0 0 115200 115200 0 115200        # UART1 = NONE (UART1 只有 RX 用不了; 释放 MSP 名额)
serial 1 1 230400 230400 0 230400        # UART2 = MSP (上位机 LicheeRV; G473 UART2 上限 230400)
#   ↑ 上位机侧: /dev/ttyS0 (UART0, 引脚 A16/A17) @230400; 与 UART2 交叉接线 A16->FC RX2/PA3, A17<-FC TX2/PA2
serial 2 64 115200 115200 0 115200       # UART3 = RX_SERIAL (CRSF 接收机)
serial 3 1 115200 115200 0 115200        # UART4 = MSP (MTF-02P 光流+测距, 内部 115200)

# --- 接收机 ---
set receiver_type = SERIAL
set serialrx_provider = CRSF

# --- MSP 接管 (RC 优先/硬件兜底) ---
# ⚠️ 位序 = AERT(内部通道号): bit0=roll bit1=pitch bit2=yaw bit3=throttle
#    (源码 rx/msp_override.c + fc/rc_controls.h: ROLL=0 PITCH=1 YAW=2 THROTTLE=3)
#    => 只接管"三轴"、油门留飞行员 = 0111 = 7 ;  (15=全接管; 11=误接管油门!)
set msp_override_channels = 7            # 只接管 roll/pitch/yaw, 油门走物理 CRSF

# --- 模式开关 (aux) —— 2026-10-12 实验20 定稿 ---
# 索引: ch5=AUX1=0  ch6=AUX2=1  ch7=AUX3=2  ch8=AUX4=3 ; modeId: ARM=0 ANGLE=1 MSP RC OVERRIDE=50
# 接管 = (ch6>1300) 或 (ch8>1300) (同 modeId 多条 aux = OR, 默认 MODE_OPERATOR_OR)
aux 0  0  0 1775 2100    # ch5 = ARM (仅解锁)
aux 1 50  1 1300 2100    # ch6>1300 -> MSP RC OVERRIDE (上位机接管)
aux 2 50  3 1300 2100    # ch8>1300 -> MSP RC OVERRIDE (与上条 OR)
aux 3  1  3  800 2100    # ch8>800  -> ANGLE
# 上位机读值(不配 FC 模式): ch6 选范式B ; ch7 定点[1300,1700)/定高[1700,2100)
# 删除旧: ch6 ANGLE[900,1300] / HORIZON[1300,1700] / ch8 override[1700,2100]

# --- 光流 + 测距 (MTF-02P 走 MSP) ---
set opflow_hardware = MSP
set rangefinder_hardware = MSP
# set opflow_scale = <实飞标定, 默认 10.5>
# set align_opflow  = <按安装朝向, 默认 CW0_DEG_FLIP>

# --- 安全 ---
set failsafe_procedure = DROP

save
